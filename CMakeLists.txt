cmake_minimum_required(VERSION 3.21)
project(velri VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)

# This first C target is the internal CPU linear kernel, not the engine API.
# One common C source is compiled into each ISA variant. Clang lowers it
# through LLVM at build time; neither LLVM nor Rust is needed at runtime.
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "The initial kernel build supports Clang and GCC")
endif()

option(VELRI_CPU_MULTIVERSION "Build optional x86 AVX2/AVX-512 variants" ON)
option(VELRI_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(VELRI_BUILD_BENCHMARK "Build the checked single-thread linear benchmark" OFF)
set(VELRI_BASE_COLUMNS 8)
set(VELRI_AVX2_COLUMNS 32)
set(VELRI_AVX512_COLUMNS 64)
set(VELRI_X86_VARIANTS 0)
set(VELRI_BASE_FLAGS)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    set(VELRI_BASE_FLAGS -march=x86-64)
    if(VELRI_CPU_MULTIVERSION)
        set(VELRI_X86_VARIANTS 1)
    endif()
endif()

function(velri_strict_target target)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
    if(VELRI_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
endfunction()

add_library(velri_cpu_linear STATIC src/backend/cpu/linear.c)
target_include_directories(velri_cpu_linear PUBLIC src/backend/cpu)
target_include_directories(velri_cpu_linear PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(velri_cpu_linear PRIVATE VELRI_X86_VARIANTS=${VELRI_X86_VARIANTS})
velri_strict_target(velri_cpu_linear)
target_compile_options(velri_cpu_linear PRIVATE ${VELRI_BASE_FLAGS} -ffp-contract=off)

# Generate narrower forms as well: small M must not allocate a full wide tile.
# The manifest serves both declarations and selection, so these cannot drift.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linear_variants.inc" "/* Generated by CMake. */\n")
function(velri_kernel_family name variant maximum pair_maximum)
    set(columns 1)
    while(columns LESS_EQUAL maximum)
        set(target velri_kernel_${name}_${columns})
        set(symbol velri_linear_${name}_c${columns})
        set(pair_columns ${columns})
        if(pair_columns GREATER pair_maximum)
            set(pair_columns ${pair_maximum})
        endif()
        add_library(${target} OBJECT src/backend/cpu/linear_kernel.c)
        target_compile_definitions(${target} PRIVATE VELRI_KERNEL=${symbol} VELRI_COLUMNS=${columns}
            VELRI_PAIR_COLUMNS=${pair_columns})
        velri_strict_target(${target})
        target_compile_options(${target} PRIVATE ${VELRI_BASE_FLAGS} ${ARGN} -ffp-contract=on)
        set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)
        target_sources(velri_cpu_linear PRIVATE $<TARGET_OBJECTS:${target}>)
        file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/linear_variants.inc"
            "VELRI_ENTRY(${symbol}, ${variant}, ${columns})\n")
        math(EXPR columns "${columns} * 2")
    endwhile()
endfunction()

velri_kernel_family(base VELRI_CPU_BASE ${VELRI_BASE_COLUMNS} ${VELRI_BASE_COLUMNS})
if(VELRI_X86_VARIANTS)
    velri_kernel_family(avx2 VELRI_CPU_AVX2 ${VELRI_AVX2_COLUMNS} 16 -mavx2 -mfma)
    velri_kernel_family(avx512 VELRI_CPU_AVX512 ${VELRI_AVX512_COLUMNS} ${VELRI_AVX512_COLUMNS} -mavx512f -mfma)
endif()

if(BUILD_TESTING OR VELRI_BUILD_BENCHMARK)
    add_library(velri_test_reference STATIC tests/cpu/reference.c)
    velri_strict_target(velri_test_reference)
    target_compile_options(velri_test_reference PRIVATE ${VELRI_BASE_FLAGS} -ffp-contract=off)
    target_include_directories(velri_test_reference PUBLIC tests/cpu)

if(BUILD_TESTING)
    add_executable(test_linear tests/cpu/test_linear.c)
    velri_strict_target(test_linear)
    target_compile_definitions(test_linear PRIVATE VELRI_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/cpu/fixtures")
    target_link_libraries(test_linear PRIVATE velri_cpu_linear velri_test_reference)
    if(UNIX)
        find_package(Threads REQUIRED)
        target_link_libraries(test_linear PRIVATE Threads::Threads)
    endif()
    add_test(NAME cpu_linear COMMAND test_linear)
endif()

    add_executable(bench_linear tools/bench_linear.c)

    target_link_libraries(bench_linear PRIVATE velri_cpu_linear velri_test_reference)
